# Werewolf (Ma Sói) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/werewolf` Discord party-game feature (Werewolf/Ma Sói: Werewolf, Villager, Seer, Doctor roles; lobby → night → day-discuss → day-vote loop) to the VCG-super-bot.

**Architecture:** Entirely in-memory, one game per Discord channel, following the exact conventions already established by `/duel-challenge-all` (`internal/discord/commands/duel_challenge_all.go`): a `sync.Mutex`-guarded `map[string]*werewolfGame` keyed by channel ID, a single goroutine driving the game through phases, Discord buttons/select-menus for interaction, DMs for private role/night actions, a private Discord thread for wolf coordination.

**Tech Stack:** Go, `discordgo` v0.29.0, existing `internal/discord/commands` package conventions (`parseOptions`, `respondError`, `respondSuccess`, `respondComponentEphemeral`, `hasAdminPermission`, `displayLabel`, `truncateLabel`, `nameTag`, `pingOnce`, `formatRemaining` — all already defined in `duel.go`/`duel_challenge_all.go`, same package, freely reusable).

## Global Constraints

- No new DB tables/migrations — everything is in-memory (spec: Persistence section).
- Game state (`cmd.games[channelID]`) must be deleted on every terminal path: game end (win), cancellation, insufficient-players-after-preflight (spec: explicit cleanup requirement added 2026-08-20).
- Any Discord message whose `Content` contains a `<@id>` mention must set `AllowedMentions` explicitly (`&discordgo.MessageAllowedMentions{}` for informational messages; `pingOnce(...)` for the one deliberate "game/round starts" ping) — never leave it nil (project-wide convention, see `internal/discord/commands/duel.go`'s `pingOnce` helper and its doc comment).
- Player count: 5-12 (`werewolfMinPlayers` / `werewolfMaxPlayers`).
- AFK removal threshold: 2 consecutive missed actions (`afkRemovalThreshold`).
- One game per channel at a time.
- Vietnamese-language user-facing strings, matching every other command in this codebase.

```go
package commands

import (
	"fmt"

const (
	werewolfMinPlayers = 5
	werewolfMaxPlayers = 12
)

// roleRatio describes how many of each special role a game of a given size gets.
type roleRatio struct {
	Werewolves int
	Seers      int
	Doctors    int
}

// rolesForPlayerCount returns the role ratio for a given player count, or an
// error if the count is outside [werewolfMinPlayers, werewolfMaxPlayers].
func rolesForPlayerCount(n int) (roleRatio, error) {
	switch {
	case n < werewolfMinPlayers:
		return roleRatio{}, fmt.Errorf("cần tối thiểu %d người, hiện có %d", werewolfMinPlayers, n)
	case n > werewolfMaxPlayers:
		return roleRatio{}, fmt.Errorf("tối đa %d người, hiện có %d", werewolfMaxPlayers, n)
	case n == 5:
		return roleRatio{Werewolves: 1}, nil
	case n == 6:
		return roleRatio{Werewolves: 1, Seers: 1}, nil
	case n <= 10:
		return roleRatio{Werewolves: 2, Seers: 1, Doctors: 1}, nil
	default: // 11-12
		return roleRatio{Werewolves: 3, Seers: 1, Doctors: 1}, nil
	}
}
```

```go
type role int

const (
	roleVillager role = iota
	roleWerewolf
	roleSeer
	roleDoctor
)

func (r role) label() string {
	switch r {
	case roleWerewolf:
		return "Ma Sói"
	case roleSeer:
		return "Tiên Tri"
	case roleDoctor:
		return "Bác Sĩ"
	default:
		return "Dân Làng"
	}
}

type player struct {
	UserID string
	Label  string
	Role   role
	Alive  bool
}

// assignRoles shuffles a copy of joinOrder and deals roles per rolesForPlayerCount.
func assignRoles(joinOrder []string, labels map[string]string) (map[string]*player, error) {
	ratio, err := rolesForPlayerCount(len(joinOrder))
	if err != nil {
		return nil, err
	}

	shuffled := make([]string, len(joinOrder))
	copy(shuffled, joinOrder)
	rand.Shuffle(len(shuffled), func(i, j int) { shuffled[i], shuffled[j] = shuffled[j], shuffled[i] })

	players := make(map[string]*player, len(shuffled))
	idx := 0
	assign := func(count int, r role) {
		for k := 0; k < count; k++ {
			id := shuffled[idx]
			idx++
			players[id] = &player{UserID: id, Label: labels[id], Role: r, Alive: true}
		}
	}
	assign(ratio.Werewolves, roleWerewolf)
	assign(ratio.Seers, roleSeer)
	assign(ratio.Doctors, roleDoctor)
	for ; idx < len(shuffled); idx++ {
		id := shuffled[idx]
		players[id] = &player{UserID: id, Label: labels[id], Role: roleVillager, Alive: true}
	}
	return players, nil
}

type gameResult int

const (
	resultOngoing gameResult = iota
	resultVillagersWin
	resultWerewolvesWin
)

// checkWinCondition: villagers win when every werewolf is dead; werewolves
// win once they're no longer outnumbered by everyone else still alive.
func checkWinCondition(players map[string]*player) gameResult {
	aliveWolves, aliveOthers := 0, 0
	for _, p := range players {
		if !p.Alive {
			continue
		}
		if p.Role == roleWerewolf {
			aliveWolves++
		} else {
			aliveOthers++
		}
	}
	switch {
	case aliveWolves == 0:
		return resultVillagersWin
	case aliveWolves >= aliveOthers:
		return resultWerewolvesWin
	default:
		return resultOngoing
	}
}
```

```go
// tallyVotes counts votes and returns every target tied for the most votes.
// isTie is true when more than one target shares the highest count — the
// caller treats that as "nobody lynched."
func tallyVotes(votes map[string]string) (topTargets []string, isTie bool) {
	counts := make(map[string]int)
	for _, target := range votes {
		counts[target]++
	}
	max := 0
	for _, c := range counts {
		if c > max {
			max = c
		}
	}
	if max == 0 {
		return nil, false
	}
	for target, c := range counts {
		if c == max {
			topTargets = append(topTargets, target)
		}
	}
	return topTargets, len(topTargets) > 1
}

const afkRemovalThreshold = 2

func shouldRemoveForAFK(strikes int) bool {
	return strikes >= afkRemovalThreshold
}
```

```go
type gamePhase int

const (
	phaseLobby gamePhase = iota
	phaseNight
	phaseDayDiscuss

// werewolfGame tracks one in-progress Werewolf game for a single channel,
// from lobby signup through to the final win check. Entirely in-memory —
// deleted from WerewolfCommand.games the instant it ends or is cancelled.
type werewolfGame struct {
	GuildID, ChannelID, StarterID string
	MessageID      string
	RevealOnDeath  bool
	Phase          gamePhase
	Round          int
	Players        map[string]*player // userID -> player
	JoinOrder      []string           // frozen once the game leaves phaseLobby
	WolfThreadID   string             // this round's wolf-coordination thread, "" between nights
	NightTarget    string             // wolves' locked-in kill target this round
	SeerActed      bool               // whether the seer has acted this round
	DoctorProtect  string             // doctor's pick this round
	DoctorLastPick string             // last round's pick — can't repeat back-to-back
	Votes          map[string]string  // voterID -> targetID, reset each day
	AFKStrikes     map[string]int     // resets to 0 on any successful action, +1 on a miss
}

// WerewolfCommand handles /werewolf and all its button/select interactions.
type WerewolfCommand struct {
	mu    sync.Mutex
	games map[string]*werewolfGame // keyed by channel ID — one game per channel
}

// NewWerewolfCommand creates a new WerewolfCommand
func NewWerewolfCommand() *WerewolfCommand {
	return &WerewolfCommand{games: make(map[string]*werewolfGame)}
}

// interactionUserID returns the acting user's ID whether the interaction
// came from a guild channel (i.Member set) or a DM (i.User set — night
// actions happen in DMs, where i.Member is always nil).
func interactionUserID(i *discordgo.InteractionCreate) string {
	if i.Member != nil && i.Member.User != nil {
		return i.Member.User.ID
	}
	if i.User != nil {
		return i.User.ID
	}
	return ""
}
```

```go
func (cmd *WerewolfCommand) handleStartGame(s *discordgo.Session, i *discordgo.InteractionCreate, channelID string) {

		return
	}
	if len(game.JoinOrder) < werewolfMinPlayers {

		return
	}
	joinOrder := make([]string, len(game.JoinOrder))
	copy(joinOrder, game.JoinOrder)
	labels := make(map[string]string, len(game.Players))
	for id, p := range game.Players {
		labels[id] = p.Label
	}
	cmd.mu.Unlock()

	_ = s.InteractionRespond(i.Interaction, &discordgo.InteractionResponse{

		Data: &discordgo.InteractionResponseData{
			Content:    "🐺 Đang chia vai trò và gửi tin nhắn riêng, chờ chút...",
			Components: []discordgo.MessageComponent{},
		},
	})

	go cmd.startPreflight(s, channelID, joinOrder, labels)
}

// dmRoleReveals DMs each listed player their assigned role, returning the
// IDs of anyone the bot couldn't reach (DMs closed/blocked).
func (cmd *WerewolfCommand) dmRoleReveals(s *discordgo.Session, players map[string]*player, ids []string) []string {
	var unreachable []string
	for _, id := range ids {
		p, ok := players[id]
		if !ok {
			continue
		}
		dm, err := s.UserChannelCreate(id)
		if err != nil {
			unreachable = append(unreachable, id)
			continue
		}
		content := fmt.Sprintf("🐺 Ván Ma Sói bắt đầu! Vai trò của bạn là: **%s**.", p.Role.label())
		if _, err := s.ChannelMessageSend(dm.ID, content); err != nil {
			unreachable = append(unreachable, id)
		}
	}
	return unreachable
}

// startPreflight assigns roles, DMs everyone their role, retries once for
// anyone unreachable (with a public grace-period notice), and drops still-
// unreachable players (recomputing roles for the smaller group) before
// kicking off round 1. Runs in its own goroutine, started from handleStartGame.
func (cmd *WerewolfCommand) startPreflight(s *discordgo.Session, channelID string, joinOrder []string, labels map[string]string) {
	players, err := assignRoles(joinOrder, labels)
	if err != nil {
		s.ChannelMessageSend(channelID, "❌ Không thể chia vai trò: "+err.Error())
		cmd.mu.Lock()
		delete(cmd.games, channelID)
		cmd.mu.Unlock()
		return
	}

	unreachable := cmd.dmRoleReveals(s, players, joinOrder)
	if len(unreachable) > 0 {
		names := make([]string, len(unreachable))
		for idx, id := range unreachable {
			names[idx] = labels[id]
		}
		s.ChannelMessageSend(channelID, fmt.Sprintf(
			"⚠️ Không gửi được tin nhắn riêng cho: %s.\nHãy bật DM từ thành viên server (Cài đặt quyền riêng tư) rồi chờ %s để bot thử lại.",
			strings.Join(names, ", "), formatRemaining(preflightGracePeriod),
		))
		time.Sleep(preflightGracePeriod)

		stillUnreachable := cmd.dmRoleReveals(s, players, unreachable)
		if len(stillUnreachable) > 0 {
			for _, id := range stillUnreachable {
				delete(players, id)
			}
			newJoinOrder := make([]string, 0, len(joinOrder))
			for _, id := range joinOrder {
				if _, ok := players[id]; ok {
					newJoinOrder = append(newJoinOrder, id)
				}
			}
			joinOrder = newJoinOrder

			if len(joinOrder) < werewolfMinPlayers {
				s.ChannelMessageSend(channelID, fmt.Sprintf("❌ Không đủ %d người sau khi loại người không nhận được tin nhắn riêng, huỷ ván.", werewolfMinPlayers))
				cmd.mu.Lock()
				delete(cmd.games, channelID)
				cmd.mu.Unlock()
				return
			}

			reassigned, err := assignRoles(joinOrder, labels)
			if err != nil {
				s.ChannelMessageSend(channelID, "❌ Không thể chia lại vai trò: "+err.Error())
				cmd.mu.Lock()
				delete(cmd.games, channelID)
				cmd.mu.Unlock()
				return
			}
			cmd.dmRoleReveals(s, reassigned, joinOrder)
			players = reassigned
		}
	}

	cmd.mu.Lock()
	game, ok := cmd.games[channelID]
	if ok {
		game.Players = players
		game.JoinOrder = joinOrder
	}
	cmd.mu.Unlock()
	if !ok {
		return
	}

	pingText, allowedMentions := pingOnce(joinOrder)
	if _, err := s.ChannelMessageSendComplex(channelID, &discordgo.MessageSend{
		Content:         fmt.Sprintf("🐺 Vai trò đã được gửi riêng cho từng người! %s", pingText),
		AllowedMentions: allowedMentions,
	}); err != nil {
		log.Error().Err(err).Msg("werewolf: failed to post role-assigned announcement")
	}

	cmd.runNightPhase(s, channelID)
}
```

```go
		p := game.Players[id]

		}
	}

func aliveNonWolves(game *werewolfGame) []*player {

		if p.Alive && p.Role != roleWerewolf {
			result = append(result, p)
		}
	}
	return result
}

func targetSelectMenu(customID, placeholder string, targets []*player) []discordgo.MessageComponent {
	options := make([]discordgo.SelectMenuOption, 0, len(targets))
	for _, p := range targets {
		options = append(options, discordgo.SelectMenuOption{
			Label: truncateLabel(p.Label),
			Value: p.UserID,
		})

	minValues := 1
	return []discordgo.MessageComponent{
		discordgo.ActionsRow{
			Components: []discordgo.MessageComponent{

					CustomID:    customID,
					Placeholder: placeholder,
					Options:     options,
					MinValues:   &minValues,
					MaxValues:   1,
				},
			},
		},
	}

// runNightPhase resets this round's night state, opens the wolf thread and
// seer/doctor DMs, waits for actions (or the timer), then resolves the night.
func (cmd *WerewolfCommand) runNightPhase(s *discordgo.Session, channelID string) {
	cmd.mu.Lock()
	game, ok := cmd.games[channelID]
	if ok {
		game.Phase = phaseNight
		game.Round++
		game.NightTarget = ""
		game.SeerActed = false
		game.DoctorProtect = ""
	}
	cmd.mu.Unlock()
	if !ok {
		return
	}

	if _, err := s.ChannelMessageSendComplex(channelID, &discordgo.MessageSend{
		Content:         fmt.Sprintf("🌙 **Đêm %d bắt đầu.** Ma Sói, Tiên Tri, Bác Sĩ đang hành động trong tin nhắn riêng...", game.Round),
		AllowedMentions: &discordgo.MessageAllowedMentions{},
	}); err != nil {
		log.Error().Err(err).Msg("werewolf: failed to post night announcement")
	}

	cmd.startWolfThread(s, game)
	cmd.dmSeerPrompt(s, game)
	cmd.dmDoctorPrompt(s, game)

	deadline := time.Now().Add(nightActionWindow)
	cmd.waitForNightActions(s, channelID, deadline)

	cmd.resolveNight(s, channelID)
}

func (cmd *WerewolfCommand) startWolfThread(s *discordgo.Session, game *werewolfGame) {
	wolves := alivePlayersOfRole(game, roleWerewolf)
	if len(wolves) == 0 {
		return
	}

	thread, err := s.ThreadStartComplex(game.ChannelID, &discordgo.ThreadStart{
		Name:                fmt.Sprintf("🐺 Sói bàn đêm %d", game.Round),
		Type:                discordgo.ChannelTypeGuildPrivateThread,
		AutoArchiveDuration: 60,
		Invitable:           false,
	})
	if err != nil {
		log.Error().Err(err).Msg("werewolf: failed to create wolf thread, falling back to independent DMs")
		cmd.dmWolvesIndependently(s, game, wolves)
		return
	}

	cmd.mu.Lock()
	game.WolfThreadID = thread.ID
	cmd.mu.Unlock()

	for _, w := range wolves {
		if err := s.ThreadMemberAdd(thread.ID, w.UserID); err != nil {
			log.Error().Err(err).Str("user_id", w.UserID).Msg("werewolf: failed to add wolf to thread")
		}
	}

	targets := aliveNonWolves(game)
	if _, err := s.ChannelMessageSendComplex(thread.ID, &discordgo.MessageSend{
		Content:    "🐺 Chọn mục tiêu để tấn công đêm nay. Lượt chọn đầu tiên sẽ được tính.",
		Components: targetSelectMenu(werewolfWolfTargetID+game.ChannelID, "Chọn mục tiêu", targets),
	}); err != nil {
		log.Error().Err(err).Msg("werewolf: failed to post wolf target select")
	}
}

// dmWolvesIndependently is the fallback when the wolf thread can't be
// created: each wolf gets the same target-picker in their own DM instead.
// The first submitted pick still stands (handleWolfTarget enforces that).
func (cmd *WerewolfCommand) dmWolvesIndependently(s *discordgo.Session, game *werewolfGame, wolves []*player) {
	targets := aliveNonWolves(game)
	for _, w := range wolves {
		dm, err := s.UserChannelCreate(w.UserID)
		if err != nil {
			log.Error().Err(err).Str("user_id", w.UserID).Msg("werewolf: failed to open DM for wolf fallback")
			continue
		}
		if _, err := s.ChannelMessageSendComplex(dm.ID, &discordgo.MessageSend{
			Content:    "🐺 Chọn mục tiêu để tấn công đêm nay.",
			Components: targetSelectMenu(werewolfWolfTargetID+game.ChannelID, "Chọn mục tiêu", targets),
		}); err != nil {
			log.Error().Err(err).Str("user_id", w.UserID).Msg("werewolf: failed to DM wolf fallback prompt")
		}
	}
}

func (cmd *WerewolfCommand) nightActionsComplete(game *werewolfGame) bool {
	if len(alivePlayersOfRole(game, roleWerewolf)) > 0 && game.NightTarget == "" {
		return false
	}
	if len(alivePlayersOfRole(game, roleSeer)) > 0 && !game.SeerActed {
		return false
	}
	if len(alivePlayersOfRole(game, roleDoctor)) > 0 && game.DoctorProtect == "" {
		return false
	}
	return true
}

// waitForNightActions blocks until every required night-actor has acted or
// the deadline passes. Posts one public reminder at the halfway point if
// the night is still waiting on someone.
func (cmd *WerewolfCommand) waitForNightActions(s *discordgo.Session, channelID string, deadline time.Time) {
	reminded := false
	for {
		cmd.mu.Lock()
		game, ok := cmd.games[channelID]
		var done bool
		if ok {
			done = cmd.nightActionsComplete(game)
		}
		cmd.mu.Unlock()
		if !ok || done || time.Now().After(deadline) {
			return
		}

		remaining := time.Until(deadline)
		if !reminded && remaining <= nightActionWindow/2 {
			reminded = true
			if _, err := s.ChannelMessageSendComplex(channelID, &discordgo.MessageSend{
				Content:         fmt.Sprintf("⏳ Còn %s để mọi người hành động đêm nay.", formatRemaining(remaining)),
				AllowedMentions: &discordgo.MessageAllowedMentions{},
			}); err != nil {
				log.Error().Err(err).Msg("werewolf: failed to post night reminder")
			}
		}

		time.Sleep(2 * time.Second)
	}
}

// resolveNight applies the kill/save outcome, records AFK strikes for
// anyone who didn't act, deletes this round's wolf thread, posts the
// result, then hands off to the win-check and (if the game continues) the
// day-discuss phase.
func (cmd *WerewolfCommand) resolveNight(s *discordgo.Session, channelID string) {
	cmd.mu.Lock()
	game, ok := cmd.games[channelID]
	if !ok {
		cmd.mu.Unlock()
		return
	}

	var killedID string
	saved := false
	if game.NightTarget != "" {
		if game.NightTarget == game.DoctorProtect {
			saved = true
		} else if target, exists := game.Players[game.NightTarget]; exists && target.Alive {
			target.Alive = false
			killedID = game.NightTarget
		}
	}

	if game.NightTarget == "" {
		for _, w := range alivePlayersOfRole(game, roleWerewolf) {
			game.AFKStrikes[w.UserID]++
		}
	} else {

		}

	var killedPlayer *player

		if _, err := s.ChannelDelete(threadID); err != nil {
			log.Error().Err(err).Msg("werewolf: failed to delete wolf thread")
```

```go
// checkAndHandleGameEnd applies AFK-strike removals, then checks the win
// condition. If the game has ended, it announces the result, reveals every
// role, deletes any remaining wolf thread, deletes the game from the
// pending map, and returns true — callers must stop processing when true.
func (cmd *WerewolfCommand) checkAndHandleGameEnd(s *discordgo.Session, channelID string) bool {
	cmd.mu.Lock()
	game, ok := cmd.games[channelID]
	if !ok {
		cmd.mu.Unlock()
		return true
	}

	var removedNames []string
	for _, id := range game.JoinOrder {
		p := game.Players[id]
		if p.Alive && shouldRemoveForAFK(game.AFKStrikes[id]) {
			p.Alive = false
			removedNames = append(removedNames, p.Label)
		}
	}

	result := checkWinCondition(game.Players)
	if result == resultOngoing {
		cmd.mu.Unlock()
		if len(removedNames) > 0 {
			cmd.announceAFKRemovals(s, channelID, removedNames)
		}
		return false
	}

	threadID := game.WolfThreadID
	delete(cmd.games, channelID)
	cmd.mu.Unlock()

	if threadID != "" {
		if _, err := s.ChannelDelete(threadID); err != nil {
			log.Error().Err(err).Msg("werewolf: failed to delete wolf thread on game end")
		}
	}

	cmd.announceGameEnd(s, game, result, removedNames)
	return true
}
```
